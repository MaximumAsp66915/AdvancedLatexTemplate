\subsection{Uncertainty}
For a result $y=f(x_1,\dots,x_n)$ with independent uncertainties $u(x_i)$,
\begin{equation}\label{eq:uncertainty}
  u(y)=\sqrt{\sum_{i=1}^{n}\Bigl(\pderiv{f}{x_i}\,u(x_i)\Bigr)^{2}}.
\end{equation}
