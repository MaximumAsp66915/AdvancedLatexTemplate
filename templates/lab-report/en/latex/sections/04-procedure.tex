\section{Procedure}

\begin{enumerate}
  \item Build the divider of \cref{fig:divider} with $R_1=\qty{1}{\kilo\ohm}$ and $R_2=\qty{2.2}{\kilo\ohm}$.
  \item Set the supply to \qty{5}{\volt} and measure $V_\mathrm{out}$.
  \item Repeat for supply voltages of \qty{2}{\volt} to \qty{10}{\volt} in steps of \qty{2}{\volt}.
  \item Connect a \qty{10}{\kilo\ohm} load across $R_2$ and measure $V_\mathrm{out}$ again.
  \item Record the instrument resolution for every reading.
\end{enumerate}
