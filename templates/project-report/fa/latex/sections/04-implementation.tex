\section{پیاده‌سازی}

هسته‌ی حلقه‌ی نمونه‌برداری در ادامه آمده است.

\begin{codebox}[title=sampler.c]{C}
#include <stdint.h>

static volatile uint32_t ticks;

void timer_isr(void) {
    ticks++;                 /* 1 Hz tick */
}

int main(void) {
    for (;;) {
        if (ticks) {
            ticks = 0;
            log_sample(read_sensor());
        }
    }
}
\end{codebox}
