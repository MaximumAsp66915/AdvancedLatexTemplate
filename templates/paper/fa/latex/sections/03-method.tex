\section{روش پیشنهادی}
\subsection{صورت‌بندی مسئله}
فرض کنید $\vect{x}\in\R^{n}$ ورودی و $f:\R^{n}\to\R$ یک تابع هدف هموار است. به دنبال
\begin{equation}\label{eq:opt}
  \vect{x}^{\star}=\argmin_{\vect{x}}\; f(\vect{x})
\end{equation}
هستیم.

\begin{theorem}\label{thm:conv}
اگر $f$ محدب و $L$-هموار باشد، کاهش گرادیان با گام $1/L$ در رابطه‌ی
$f(\vect{x}_k)-f^{\star}\le L\norm{\vect{x}_0-\vect{x}^{\star}}^{2}/(2k)$ صدق می‌کند.
\end{theorem}

\subsection{الگوریتم}
\Cref{eq:opt} با تکرار $\vect{x}_{k+1}=\vect{x}_k-\eta\nabla f(\vect{x}_k)$ حل می‌شود.
