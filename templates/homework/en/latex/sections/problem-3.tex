\begin{problem}[Units][5]
A resistor of \qty{4.7}{\kilo\ohm} carries \qty{2.5}{\milli\ampere}. What power does it dissipate?
\end{problem}

\begin{solution}
$P = I^2R = (\qty{2.5}{\milli\ampere})^2\cdot\qty{4.7}{\kilo\ohm} \approx \qty{29.4}{\milli\watt}$.
\end{solution}

\begin{problem}[Proof][10]
Show that $\sqrt{2}$ is irrational.
\end{problem}

\begin{proof}
Suppose $\sqrt2=p/q$ in lowest terms. Then $p^2=2q^2$, so $p$ is even, say $p=2k$. Then $q^2=2k^2$ and $q$ is even as well, a contradiction.
\end{proof}
