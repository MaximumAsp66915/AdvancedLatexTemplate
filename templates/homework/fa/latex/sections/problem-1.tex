\section{دستگاه‌های خطی}

\begin{problem}[مقادیر ویژه][۱۰]
مقادیر ویژه و بردارهای ویژه‌ی ماتریس زیر را بیابید:
\[
  \mat{A}=\begin{bmatrix} 2 & 1\\ 1 & 2\end{bmatrix}.
\]
\end{problem}

\begin{solution}
چندجمله‌ای مشخصه برابر است با
\[
  \det(\mat{A}-\lambda\mat{I}) = (2-\lambda)^2-1 = (\lambda-1)(\lambda-3)\text{،}
\]
پس $\lambda_1=1$ و $\lambda_2=3$. بردارهای ویژه‌ی متناظر
$\vect{v}_1=\trans{[1,\,-1]}$ و $\vect{v}_2=\trans{[1,\,1]}$ هستند.

\begin{resultbox}[پاسخ]
$\lambda_1=1$ ،\quad $\lambda_2=3$.
\end{resultbox}
\end{solution}

\begin{problem}[معین مثبت بودن][۵]
آیا $\mat{A}$ معین مثبت است؟ پاسخ خود را توجیه کنید.
\end{problem}

\begin{solution}
بله. هر دو مقدار ویژه مثبت‌اند، پس برای هر $\vect{x}\neq\vect{0}$ داریم $\trans{\vect{x}}\mat{A}\vect{x}>0$.
\end{solution}
