% ---- listings: give every glyph its own color ---------------------------------------------
% Ordinary punctuation has no style of its own (\lst@thestyle is \relax), so it inherits the
% surrounding text color. Inside a breakable box (solution, problem, codebox ...) that color
% is black again on every page after the first, so characters such as & # % ^ ~ | `  turned
% black on a dark theme. The OutputOther hook runs just before such a token is typeset:
% give it the code color. `)' gets the bracket color: with breaklines=true listings
% re-defines `)' itself (to allow a break after it), which bypasses any literate rule.
\def\atl@cparen{)}
\lst@AddToHook{OutputOther}{%
  \ifx\lst@thestyle\relax
    \edef\atl@tok{\the\lst@token}%
    \ifx\atl@tok\atl@cparen
      \def\lst@thestyle{\color{atl-lst-br}}%
    \else
      \def\lst@thestyle{\color{atl-lst-fg}}%
    \fi
  \fi}
